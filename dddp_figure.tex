\begin{figure*}[t]
\centering
\begin{tikzpicture}[
  font=\small,
  >={Stealth[length=2mm]},
  win/.style={draw, thick, rounded corners, minimum width=23mm,
              minimum height=11mm, align=center, fill=blue!6},
  emphwin/.style={draw, very thick, rounded corners, minimum width=23mm,
                  minimum height=11mm, align=center, fill=orange!15,
                  draw=orange!75!black},
  fwd/.style={-{Stealth[length=2mm]}, thick, blue!55!black},
  bwd/.style={-{Stealth[length=2mm]}, thick, red!70!black, dashed},
  hdr/.style={font=\small\bfseries},
]
\def\cx{4.0}
\def\ry{3.2}

% ---------- lattice nodes ----------
% top row: iteration k-1
\node[win] (a) at (0,0)       {$[\mathrm{SP}_{i-1}^{(k-1)}]$};
\node[win] (b) at (\cx,0)     {$[\mathrm{SP}_{i}^{(k-1)}]$};
\node[win] (c) at (2*\cx,0)   {$[\mathrm{SP}_{i+1}^{(k-1)}]$};
% bottom row: iteration k
\node[win]     (d) at (0,-\ry)     {$[\mathrm{SP}_{i-1}^{(k)}]$};
\node[emphwin] (e) at (\cx,-\ry)   {$[\mathrm{SP}_{i}^{(k)}]$};
\node[win]     (f) at (2*\cx,-\ry) {$[\mathrm{SP}_{i+1}^{(k)}]$};

% ---------- headers / axis labels ----------
\node[hdr] at (0,1.15)     {$\mathcal{W}_{i-1}$};
\node[hdr] at (\cx,1.15)   {$\mathcal{W}_{i}$};
\node[hdr] at (2*\cx,1.15) {$\mathcal{W}_{i+1}$};
\node[hdr, left=8mm of a] {iter.\ $k{-}1$};
\node[hdr, left=8mm of d] {iter.\ $k$};

% ---------- forward trial SOC: down-right, solid ----------
\draw[fwd] (a) to
   node[sloped, above, pos=0.34, font=\footnotesize]
   {$\hat{\bm{x}}_{i-1}^{(k)}{=}\bm{x}_{i-1}^{(k-1)}$} (e);
\draw[fwd, opacity=0.5] (b) to
   node[sloped, above, pos=0.30, font=\footnotesize, opacity=1]
   {$\hat{\bm{x}}_{i}^{(k)}{=}\bm{x}_{i}^{(k-1)}$} (f);

% ---------- backward Benders cut: down-left, dashed ----------
\draw[bwd] (c) to
   node[sloped, below, pos=0.34, font=\footnotesize]
   {$(\alpha_i^{(k-1)},\bm{\beta}_i^{(k-1)})$ on $\theta_i$} (e);
\draw[bwd, opacity=0.5] (b) to
   node[sloped, below, pos=0.30, font=\footnotesize, opacity=1]
   {$(\alpha_{i-1}^{(k-1)},\bm{\beta}_{i-1}^{(k-1)})$ on $\theta_{i-1}$} (d);

% ---------- callout for the highlighted window ----------
\node[align=center, font=\footnotesize, below=9mm of e, text width=0.92\linewidth] (note)
{Window $i$ at iteration $k$ consumes the
 \textcolor{blue!55!black}{forward trial SOC} from window $i{-}1$ and the
 \textcolor{red!70!black}{backward cut} from window $i{+}1$, both produced
 at iteration $k{-}1$; it then emits its own forward SOC and backward cut
 to iteration $k{+}1$. Every $[\mathrm{SP}_i^{(k)}]$ therefore depends only
 on iteration-$(k{-}1)$ data, so all windows solve in parallel.};

% ---------- legend ----------
\coordinate (lg) at ([yshift=-4mm]note.south west);
\draw[fwd] (lg) -- ++(7mm,0)
   node[right, black, font=\footnotesize]{forward trial SOC (down-right)};
\draw[bwd] ([yshift=-4.5mm]lg) -- ++(7mm,0)
   node[right, black, font=\footnotesize]{backward Benders cut (down-left)};

\end{tikzpicture}
\caption{Space--time view of the parallel DDDP recursion over two
successive outer iterations ($k{-}1$ and $k$) for three adjacent windows
($i{-}1,\,i,\,i{+}1$). Forward trial SOCs travel \emph{down-right}
(solid): the terminal SOC $\bm{x}_{i-1}^{(k-1)}$ of window $i{-}1$ becomes
the fixed trial $\hat{\bm{x}}_{i-1}^{(k)}$ of window $i$. Backward
optimality cuts travel \emph{down-left} (dashed): the cut
$(\alpha_i^{(k-1)},\bm{\beta}_i^{(k-1)})$ generated by window $i{+}1$
bounds the value-to-go $\theta_i$ in window $i$. Because every
sub-problem $[\mathrm{SP}_i^{(k)}]$ uses only iteration-$(k{-}1)$ data,
the $P$ windows are solved simultaneously.}
\label{fig:dddp}
\end{figure*}
